// ─────────────────────────────────────────────────────────────
// 골격 — {{...}} 네 곳만 바꾸면 된다
// ─────────────────────────────────────────────────────────────
// {{FOLDER}}  SharePoint DocLib 하위 폴더명. 슬래시 없이
// {{PREFIX}}  파일명 앞머리
// {{COL}}     내보낼 컬렉션
// {{FIELDS}}  CSV 열 매핑

=Set(gblExportFolder,"{{FOLDER}}");
Set(gblExportFileName,"{{PREFIX}}_" & Text(Now(),"yyyymmdd_hhmmss") & ".csv");
If(CountRows({{COL}}) = 0,
    Notify("내보낼 데이터가 없습니다.", NotificationType.Warning),
    '자산관리-목록CSV내보내기'.Run(
        gblExportFileName,
        JSON(ForAll({{COL}} As X, { {{FIELDS}} }), JSONFormat.Compact),
        "/{{FOLDER}}"
    );
    Set(gblExportDone, true)
)

// gblExportFolder 에는 슬래시를 넣지 않는다. 화면 표시와 폴더 URL 조합에 쓰기 때문이다.
// .Run() 의 3번째 인자에만 슬래시를 붙인다. 플로우가 "/DocLib" 뒤에 그대로 이어 붙인다.
